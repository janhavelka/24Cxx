if(COMMAND idf_component_register)
  idf_component_register(SRCS "src/EEPROM24Cxx.cpp" "src/BlockingMemory.cpp" INCLUDE_DIRS "include")
  target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
else()
  cmake_minimum_required(VERSION 3.16)
  project(EEPROM24Cxx LANGUAGES CXX)
  add_library(EEPROM24Cxx src/EEPROM24Cxx.cpp src/BlockingMemory.cpp)
  target_include_directories(EEPROM24Cxx PUBLIC include)
  target_compile_features(EEPROM24Cxx PUBLIC cxx_std_17)
  option(EEPROM24CXX_BUILD_TESTS "Build native regression tests" OFF)
  if(EEPROM24CXX_BUILD_TESTS)
    enable_testing()
    add_executable(eeprom24cxx_tests test/test_core/test_core.cpp)
    target_link_libraries(eeprom24cxx_tests PRIVATE EEPROM24Cxx)
    add_test(NAME core COMMAND eeprom24cxx_tests)
    add_executable(eeprom24cxx_cli_tests test/test_cli/test_cli.cpp examples/common/Eeprom24CxxCli.cpp)
    target_include_directories(eeprom24cxx_cli_tests PRIVATE examples/common)
    target_link_libraries(eeprom24cxx_cli_tests PRIVATE EEPROM24Cxx)
    add_test(NAME cli COMMAND eeprom24cxx_cli_tests)
    add_executable(eeprom24cxx_transport_tests test/test_transport/test_transport.cpp)
    target_include_directories(eeprom24cxx_transport_tests PRIVATE examples/common)
    target_link_libraries(eeprom24cxx_transport_tests PRIVATE EEPROM24Cxx)
    add_test(NAME transport COMMAND eeprom24cxx_transport_tests)
    add_executable(eeprom24cxx_helper_tests test/test_helpers/test_helpers.cpp)
    target_link_libraries(eeprom24cxx_helper_tests PRIVATE EEPROM24Cxx)
    add_test(NAME helpers COMMAND eeprom24cxx_helper_tests)
    add_executable(eeprom24cxx_diagnostic_tests test/test_diagnostics/test_diagnostics.cpp examples/common/Eeprom24CxxCli.cpp)
    target_include_directories(eeprom24cxx_diagnostic_tests PRIVATE examples/common)
    target_link_libraries(eeprom24cxx_diagnostic_tests PRIVATE EEPROM24Cxx)
    add_test(NAME diagnostics COMMAND eeprom24cxx_diagnostic_tests)
    foreach(suite IN ITEMS detection console)
      add_executable(eeprom24cxx_${suite}_tests test/test_${suite}/test_${suite}.cpp examples/common/Eeprom24CxxCli.cpp)
      target_include_directories(eeprom24cxx_${suite}_tests PRIVATE examples/common)
      target_link_libraries(eeprom24cxx_${suite}_tests PRIVATE EEPROM24Cxx)
      add_test(NAME ${suite} COMMAND eeprom24cxx_${suite}_tests)
    endforeach()
    find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
    add_test(NAME hil_runner COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tools -p "test_hil_*.py")
    set_tests_properties(hil_runner PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(eeprom24cxx_cli_host test/cli_host/main.cpp examples/common/Eeprom24CxxCli.cpp)
    target_include_directories(eeprom24cxx_cli_host PRIVATE examples/common)
    target_link_libraries(eeprom24cxx_cli_host PRIVATE EEPROM24Cxx)
    add_test(NAME cli_transcript COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_cli_transcript.py" $<TARGET_FILE:eeprom24cxx_cli_host>)
  endif()
endif()
